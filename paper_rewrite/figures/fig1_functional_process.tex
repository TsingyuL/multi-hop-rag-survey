\definecolor{initialblue}{RGB}{74,112,148}
\definecolor{followgreen}{RGB}{83,128,116}
\definecolor{integrateorange}{RGB}{168,119,71}

\node[endpoint] (task) {\textbf{Task}\\$q$};
\node[stage=initialblue, right=of task] (initial) {\textbf{Initial Retrieval}\\Find entry evidence\\$q\rightarrow E_1$};
\node[stage=followgreen, right=of initial] (follow) {\textbf{Follow-up Retrieval}\\Resolve the next need\\$(q,E_{1:t})\rightarrow E_{t+1}$};
\node[stage=integrateorange, right=of follow] (integrate) {\textbf{Evidence Integration}\\Select, preserve, combine\\$E_{1:T}\rightarrow C$};

\draw[arrow] (task) -- (initial);
\draw[arrow] (initial) -- (follow);
\draw[arrow] (follow) -- (integrate);
\draw[arrow] (integrate) -- (output);

\draw[taskloop] (follow.south east) to[out=-65,in=-115,looseness=5.0]
  node[below,note]{update state, search again, or stop} (follow.south west);

\node[feedback, below=18mm of follow, xshift=14mm] (learning) {
\textbf{Feedback Learning}: diagnose completed trajectories and update persistent

$(\tau_i,f_i,M_i)\rightarrow (M_{i+1},\pi_{i+1})$};

\draw[learnloop] (output.south) to[out=-90,in=15]
  node[right,note]{evaluation} (learning.east);
\draw[learnloop] (learning.west) to[out=170,in=-90]
  node[below left,note]{improve future tasks} (initial.south);
